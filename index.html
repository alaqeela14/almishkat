function buildLatestUpdates() { const items = []; const articles = mishkatArticles[currentLang] || mishkatArticles.ar; const books = mishkatBooks[currentLang] || mishkatBooks.ar; articles.forEach((a, i) => items.push({ type: 'article', typeLabel: 'مقال', typeIcon: 'fa-newspaper', title: a.title, excerpt: a.excerpt, author: a.author, date: a.date, image: a.image, index: i, catKey: a.catKey, sortDate: a.date ? new Date(a.date).getTime() : 0 })); mishkatNews.forEach((n, i) => items.push({ type: 'news', typeLabel: n.type === 'video' ? 'خبر مرئي' : 'خبر', typeIcon: n.type === 'video' ? 'fa-video' : 'fa-newspaper', title: n.title, excerpt: n.excerpt || '', author: '', date: n.date, image: n.image || '', index: i, sortDate: n.date ? new Date(n.date).getTime() : 0 })); books.forEach((b, i) => items.push({ type: 'book', typeLabel: 'كتاب', typeIcon: 'fa-book', title: b.title, excerpt: b.description || '', author: b.author, date: b.date, image: b.cover || '', index: i, sortDate: b.date ? new Date(b.date).getTime() : 0 })); mishkatAudios.forEach((a, i) => items.push({ type: 'audio', typeLabel: 'صوتي', typeIcon: 'fa-headphones', title: a.title, excerpt: a.description || '', author: a.speaker || '', date: a.date, image: '', index: i, sortDate: a.date ? new Date(a.date).getTime() : 0 })); mishkatVideos.forEach((v, i) => items.push({ type: 'video', typeLabel: 'مرئي', typeIcon: 'fa-video', title: v.title, excerpt: v.description || '', author: v.channel || '', date: v.date, image: '', index: i, sortDate: v.date ? new Date(v.date).getTime() : 0 })); items.sort((a, b) => b.sortDate - a.sortDate); return items.slice(0, 14); } function openLatestItem(type, index) { if (type === 'article') openArticle(index); else if (type === 'news') openNews(index); else if (type === 'book') { showView('library'); setTimeout(() => { const el = document.querySelectorAll('.book-card')[index]; if (el) el.scrollIntoView({ behavior:'smooth', block:'center' }); }, 300); } else if (type === 'audio') { showView('audios'); setTimeout(() => { const el = document.querySelectorAll('#audiosList .audio-card')[index]; if (el) el.scrollIntoView({ behavior:'smooth', block:'center' }); }, 300); } else if (type === 'video') { showView('videos'); setTimeout(() => { const el = document.querySelectorAll('#videosList .video-card')[index]; if (el) el.scrollIntoView({ behavior:'smooth', block:'center' }); }, 300); } } function renderArticles(filterCat) { const grid = document.getElementById('articlesGrid'); const empty = document.getElementById('articlesEmpty'); const t = translations[currentLang]; const articles = mishkatArticles[currentLang] || mishkatArticles.ar; const isFiltered = (filterCat && filterCat !== 'all'); if (isFiltered) document.body.classList.add('extras-hidden'); else document.body.classList.remove('extras-hidden'); if (isFiltered) { let filtered = articles.filter(a => a.catKey === filterCat); if (filtered.length === 0) { grid.innerHTML = ''; empty.style.display = 'block'; return; } empty.style.display = 'none'; grid.innerHTML = filtered.map(a => { const idx = articles.indexOf(a); return `
${t[a.catKey] || ''}

${a.title}

${a.excerpt}

${a.author} ${a.date}
`; }).join(''); } else { const items = buildLatestUpdates(); if (items.length === 0) { grid.innerHTML = ''; empty.style.display = 'block'; return; } empty.style.display = 'none'; grid.innerHTML = items.map(it => { return `
${it.typeLabel}

${it.title}

${it.excerpt}

${it.author || 'مشكاة'} ${it.date || ''}
`; }).join(''); } }
